Plugin source — BloodMoonEvent.java 
public final class BloodMoonEvent implements Listener {
  private final RealmCore plugin;
  private final BossBar bar = Bukkit.createBossBar("Blood Moon", BarColor.RED, BarStyle.SEGMENTED_10);
  private int left;

  public void start(World world, int goal) {
    left = goal;
    world.setTime(13_000);
    Bukkit.getOnlinePlayers().forEach(bar::addPlayer);
    Msg.broadcast("&4[EVENT] &cThe Blood Moon rises!");
    plugin.scheduler().every(40, () -> spawnWave(world));
  }

  @EventHandler
  public void onKill(EntityDeathEvent e) {
    if (!(e.getEntity() instanceof Zombie) || left <= 0) return;
    bar.setProgress(--left / (double) 8);
    if (left == 0) reward(e.getEntity().getKiller());
  }

  private void reward(Player p) {
    plugin.economy().depositPlayer(p, 500);
    p.getInventory().addItem(Crates.key("legend"));
    p.getWorld().spawn(p.getLocation(), Firework.class);
    bar.removeAll();
  }
}
Plugin source — DailyCommand.java 
public final class DailyCommand implements CommandExecutor {
  private final Economy eco;
  private final Map<UUID, Long> claimed = new HashMap<>();

  @Override
  public boolean onCommand(CommandSender s, Command c, String l, String[] a) {
    if (!(s instanceof Player p)) return true;
    long last = claimed.getOrDefault(p.getUniqueId(), 0L);
    if (System.currentTimeMillis() - last < 86_400_000L) {
      p.sendMessage(Msg.color("&cCome back tomorrow!"));
      return true;
    }
    eco.depositPlayer(p, 250);
    claimed.put(p.getUniqueId(), System.currentTimeMillis());
    p.getWorld().spawnParticle(Particle.TOTEM_OF_UNDYING, p.getLocation(), 60);
    p.sendMessage(Msg.color("&a+250 coins &7— daily reward claimed"));
    return true;
  }
}
