public final class DailyCommand implements CommandExecutor {
  private final Economy eco;
  private final Map<UUID, Long> claimed = new HashMap<>();

  @Override
  public boolean onCommand(CommandSender s, Command c, String l, String[] a) {
    if (!(s instanceof Player p)) return true;
    long last = claimed.getOrDefault(p.getUniqueId(), 0L);
    if (System.currentTimeMillis() - last < 86_400_000L) {
      p.sendMessage(Msg.color("&cCome back tomorrow!"));
      return true;
    }
    eco.depositPlayer(p, 250);
    claimed.put(p.getUniqueId(), System.currentTimeMillis());
    p.getWorld().spawnParticle(Particle.TOTEM_OF_UNDYING, p.getLocation(), 60);
    p.sendMessage(Msg.color("&a+250 coins &7— daily reward claimed"));
    return true;
  }
}
