          Telegram shop bot in 20 linesNode.js · grammY · inline keyboards

      import { Bot, InlineKeyboard } from 'grammy';

const bot = new Bot(process.env.BOT_TOKEN);
const products = [{ id: 'lamp', name: 'Neon Katana Lamp', price: 59 }];

bot.command('start', ctx => ctx.reply('Welcome to Kage Shop!', {
  reply_markup: new InlineKeyboard().text('Catalog', 'catalog')
}));

bot.callbackQuery('catalog', async ctx => {
  const kb = new InlineKeyboard();
  products.forEach(p => kb.text(`${p.name} — ${p.price} USDT`, `buy:${p.id}`).row());
  await ctx.editMessageText('Pick a product:', { reply_markup: kb });
});

bot.callbackQuery(/^buy:(.+)$/, async ctx => {
  const p = products.find(x => x.id === ctx.match[1]);
  await ctx.reply(`Send ${p.price} USDT (TRC20). Payment is detected automatically.`);
  await ctx.answerCallbackQuery();
});

bot.start();
      
        
      
        
      
          Chrome extension that highlights keywordsManifest V3 · content script · storage

      // manifest.json
{
  "manifest_version": 3,
  "name": "FocusKit",
  "version": "1.0.0",
  "permissions": ["storage"],
  "content_scripts": [{ "matches": ["<all_urls>"], "js": ["content.js"] }]
}

// content.js
chrome.storage.sync.get({ words: ['crypto', 'agents'] }, ({ words }) => {
  const re = new RegExp(`\\b(${words.join('|')})`, 'gi');
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  const hits = [];
  while (walker.nextNode()) if (re.test(walker.currentNode.data)) hits.push(walker.currentNode);
  hits.forEach(node => {
    const span = document.createElement('span');
    span.innerHTML = node.data.replace(re, '<mark>$1</mark>');
    node.replaceWith(span);
  });
});

          Verify a crypto payment webhookNode.js · Express · HMAC-SHA256 · timing-safe compare

      import crypto from 'node:crypto';
import express from 'express';

const app = express();

app.post('/webhooks/crypto', express.raw({ type: 'application/json' }), (req, res) => {
  const sig = req.get('X-Signature') || '';
  const expected = 'sha256=' + crypto
    .createHmac('sha256', process.env.WEBHOOK_SECRET)
    .update(req.body).digest('hex');

  const valid = sig.length === expected.length &&
    crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected));
  if (!valid) return res.sendStatus(401);

  const event = JSON.parse(req.body);
  if (event.event === 'payment.confirmed' && event.confirmations >= 3) {
    fulfillOrder(event.invoice); // idempotent: safe if the webhook is retried
  }
  res.sendStatus(200);
});
